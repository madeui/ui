## Install

```bash
npx shadcn@latest add @ui-lib/tooltip
```

## Usage

```tsx
import {
  Tooltip,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from '@/components/ui/tooltip';

// Wrap your app (or subtree) in <TooltipProvider> once.
```
